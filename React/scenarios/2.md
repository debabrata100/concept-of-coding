# The complaint: "Our live price ticker sometimes shows stale prices after users switch symbols, and DevTools shows memory usage climbing the longer the tab is open."

function PriceTicker({ symbol }) {
  const [price, setPrice] = useState(null);
  const [history, setHistory] = useState([]);
  const [alertThreshold, setAlertThreshold] = useState(100);

  useEffect(() => {
    const socket = new WebSocket(`wss://api.example.com/prices/${symbol}`);

    socket.onmessage = (event) => {
      const newPrice = JSON.parse(event.data).price;
      setPrice(newPrice);
      setHistory([...history, newPrice]);

      if (newPrice > alertThreshold) {
        console.warn(`${symbol} crossed ${alertThreshold}: ${newPrice}`);
      }
    };
  }, [symbol]);

  return (
    <div>
      <h2>{symbol}: ${price}</h2>
      <input
        type="number"
        value={alertThreshold}
        onChange={e => setAlertThreshold(Number(e.target.value))}
      />
      <p>History: {history.length} points</p>
    </div>
  );
}

// Parent usage:
<PriceTicker symbol={selectedSymbol} />  // selectedSymbol changes when user clicks a different stock

Reported symptoms:

- When the user switches from AAPL to TSLA, they sometimes still see AAPL price updates flickering in for a few seconds before TSLA takes over.
- The alert warning console.warn fires against the old threshold value even after the user has changed the threshold in the input.
- Memory grows steadily; Chrome's memory profiler shows an increasing number of live WebSocket connections after each symbol switch.
- The history.length sometimes appears to reset or skip numbers when prices arrive rapidly.

useEffect(() => {
  const socket = new WebSocket(`wss://api.example.com/prices/${symbol}`);

  socket.onmessage = (event) => {
    const newPrice = JSON.parse(event.data).price;
    setPrice(newPrice);
    setHistory(prev => [...prev, newPrice]);      // functional updater — fixes Bug 4

    if (newPrice > alertThresholdRef.current) {   // ref read — fixes Bug 2
      console.warn(`${symbol} crossed ${alertThresholdRef.current}: ${newPrice}`);
    }
  };

  return () => socket.close();                    // fixes Bugs 1 & 3
}, [symbol]);
