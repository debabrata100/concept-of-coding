# You ahve to render a component when an external variable changes

```
import { useSyncExternalStore } from "react";

let score = 0;
let listener = null;

function getSnapshot() {
  return score;
}

function subscribe(callback) {
  listener = callback;
}

function App() {
  const newScore = useSyncExternalStore(subscribe, getSnapshot);

  function updateScore(newScore) {
    score = newScore;
    if (listener) {
      listener();
    }
  }

  return (
    <div>
      <h1>This is your App</h1>
      <p>Render a component by updating a normal variable</p>
      <p>Score: {newScore}</p>
      <button onClick={() => updateScore(score + 1)}>Increment Score</button>
    </div>
  );
}

export default App;
```
